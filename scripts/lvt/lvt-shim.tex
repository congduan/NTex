% lvt-shim.tex —— 用 plain 承载 l3build 官方测试（.lvt）的最小垫片
%
% ## 为什么需要它
%
% l3kernel 官方测试（266 个 .lvt / 其中 208 个 l3kernel）是**权威分母**：
% 每个 .lvt 配一个 .tlg（期望转录，gold standard）。但 .lvt 头部是
% `\documentclass{minimal}` + `\input{regression-test}` —— 需要 LaTeX 内核，
% 而 LaTeX 内核又依赖 expl3（循环依赖）。本垫片用 plain 提供那几个 LaTeX 符号，
% 使测试体（纯 expl3）可在 NTex 上直接跑。
%
% ## 用法
%
%     ntex-dvi lvt-shim.tex "\def\LVTFILE{<name>}\input{lvt-shim}"
%
% 垫片做三件事：
%   1. 提供 LaTeX 最小符号（\documentclass/\begin/\end/\makeatletter/@catcode）
%   2. 载入官方 regression-test.tex（提供 \TEST/\TYPE/\START/\END）
%   3. \input 目标 .lvt（其 `\documentclass{minimal}` 已被垫片吞掉）
%
% ## 已知边界
%
% 垫片不实现 LaTeX 内核功能；.lvt 体若真用 LaTeX 命令（少数用 `\text`/
% 环境等）会报差异——那**本身就是有价值的信息**（说明该测试超出 expl3 范围）。

\catcode`\@=11

% ── LaTeX 最小符号 ──────────────────────────────────────────────
% \documentclass[opts]{class}：吃可选参数 + 必选参数，无操作
% （旧实现用 \@gobbleopt + \futurelet 链，疑似在 LVT 上下文残留状态
%  污染后续 cs 解析 —— 见 docs/latex-feasibility.md §A1.sedecies。
%  LVT 用例里没有可选参数，直接简化。）
% \@ifnextchar 不存在于 plain，用 \futurelet 单层判 `[`。
\def\documentclass{\futurelet\@let@token\@docclass}
\def\@docclass{\ifx\@let@token[\expandafter\@docclass@opt\else\expandafter\@docclass@arg\fi}
\def\@docclass@opt[#1]{\@docclass@arg}
\def\@docclass@arg#1{}

% \begin{document}/\end{document}：无操作（内容是测试体本身）
%
% ⚠ **不可** `\def\end#1{}` 遮蔽原语 `\end`：regression-test.tex 的
% `\let\@@@end\end` 要绑的是**原语** `\end`（作业终止）。用 `\begin` 单独
% 吞掉 `{document}` 即可（`\end{document}` 里的 `\end` 交给原语，但原语
% `\end` 后跟 `{document}` 会误当正文——故用 `\AtEndDocument` 式惰性处理：
% 把 `\end` 包成"若下一 token 是 `{` 则吞参数组，否则转原语"）。
\def\begin#1{}
% `\end` 保持**原语**，不做任何包装。
% （曾用 `\futurelet` 包装以吞 `{document}`，但 regression-test.tex 的
%  `\END` 宏体末尾是裸 `\@@@end`（后面无 `{`），包装器毫无必要；
%  过度设计反而引入状态问题——见 docs/latex-feasibility.md §A1.septendecies。）
\let\@@ntex@end@prim\end

\def\usepackage#1{}
\def\RequirePackage#1{}
% ── LaTeX 钩子：`\AtBeginDocument` / `\AtEndDocument` ────────────────────
%
% ⚠ 必须提供（2026-09-11 实测，价值极高）：`expl3-code.tex` L13157 用
%   `\cs_if_exist:NT \@filelist { \AtBeginDocument { … } }` 包裹一段代码块。
%   这两个宏是 **LaTeX 内核**命令，plain 无。缺失时：
%     - 报 `! Undefined control sequence. \AtBeginDocument`
%     - 其后 `}` 配对全乱 → `Argument of \__file_tmp:w has an extra }` 级联
%     - **载入在 L25851 崩溃，产生 3,114 个错误**（实测）
%   补上后 expl3-code.tex（40,266 行）**零错误完整载入**。
%
%   ⇒ 这 3,114 个错误**全是脚手架缺陷**，不是引擎缺陷。此前多轮
%     「引擎 bug」误判源于此（与 2026-09-11 那轮 harness 载入两次同类）。
%
%   语义：真实 LaTeX 里它们把代码延迟到 `\begin{document}` 执行。垫片无此
%   时机概念 → 直接执行（expl3 的 `\@filelist` 分支在 plain 下本就不该走，
%   且该段仅做文件清单记录，副作用可忽略）。
%   ⚠ 参数必须吃下：用不带 `#1` 的 `\def` 会让 `{…}` 组成为裸组，
%     组内 `\catcode` 变化泄漏；故必须吞掉实参组。
\def\AtBeginDocument#1{}
\def\AtEndDocument#1{}
\def\makeatletter{\catcode`\@=11 }
\def\makeatother{\catcode`\@=12 }
\def\ProvidesFile#1{}
\def\NeedsTeXFormat#1{}
\def\ProvidesPackage#1{}
\def\ProcessOptions{\relax}
% ── expl3 调试设施（l3debug.dtx L108 `\debug_on:n`）────────────────────────
% `.lvt` 头部在 `\ExplSyntaxOn` 后调 `\debug_on:n { check-declarations ,
% deprecation , log-functions }`——这是 **expl3 kernel 函数**，plain 无。
% 不提供则报 Undefined → 级联污染（实测 m3basics001 首个错误即 `\debug_on:n`，
% 其后测试体全部错位）。调试开关对输出**无影响**（只加检查），故空实现即可。
\def\debug_on:n#1{}
\def\debug_off:n#1{}
% ── `\ExplSyntaxOn` / `\ExplSyntaxOff`（LaTeX 内核 / l3bootstrap 提供）──────
% 官方定义见 l3kernel/l3bootstrap.dtx L317+：开启时把
%   tab(9)/space(32) → ignore、双引号(34)/竖线(124) → other、
%   冒号(58)/下划线(95) → **letter**、脱字符(94) → math_superscript、
%   波浪号(126) → space；`\endlinechar` 置 32。
% 关掉时恢复原值。
%
% ⚠ 这两个宏**必须存在**，否则 `_`/`:` 保持 catcode 8/12，`\tl_new:N \l_A_tl`
% 里的 `_` 被当数学下标 → `! 双重下标`（实测 46 例 CRASH 的真根因，
% 见 docs/expl3-lvt-scoreboard.md）。plain 无此宏，须 shim 提供。
% ⚠ 变量名**不可含 `_`**：plain 下 `_` 是 catcode 8（下标），`\l__lvt@x` 会被
% 切成 `\l` + 字面 `__lvt@x` —— 定义处与使用处解析成不同 cs，静默失效
% （实测：`\ExplSyntaxOn` 被调用但不生效，`_` 仍为 cat 8，导致 187 例测试体
% 里 `\cs_if_exist_use:N` 等名字渲染成 `cs\~if\~exist\~use`）。
% shim 载入时 `_` 尚未变字母，故内部名一律用 `@`（`@` 在 shim 已设 cat 11）。
\chardef\lvt@expl@bool=0
\def\ExplSyntaxOn{%
  \ifnum\lvt@expl@bool=0
    \catcode`\^^I=9 \catcode`\ =9 \catcode`\"=12
    \catcode`\:=11 \catcode`\_=11 \catcode`\^=7
    \catcode`\|=12
    % ⚠ `~`(126) 在 expl3 语法下是 **catcode 10（space）**，不是 13！
    % 官方 l3bootstrap.dtx L345 `\char_set_catcode_space:n { 126 } % tilde`。
    % 写成 13 使 `.lvt` 测试标题里的 `~`（作空格用，如 `cs~if~exist~use`）
    % 被当 active char 展开成 `\~`（重音宏），187 例测试标题全渲染错。
    \catcode`\~=10
    \endlinechar=32
    \chardef\lvt@expl@bool=1
  \fi}
\def\ExplSyntaxOff{%
  \ifnum\lvt@expl@bool=1
    \catcode`\^^I=10 \catcode`\ =10 \catcode`\"=12
    \catcode`\:=12 \catcode`\_=8 \catcode`\^=7
    \catcode`\|=12 \catcode`\~=13
    \endlinechar=13
    \chardef\lvt@expl@bool=0
  \fi}
% ⚠ 不可 `\let\@undefined\relax`！LaTeX **内核里 `\@undefined` 本身是未定义的**
% 名字，`\ifx\X\@undefined` 判「X 是否未定义」靠的是 **undefined vs undefined
% 同类比较**（tex.web：eqtb 空槽 vs 空槽 → 类型同 → 真）。
% 若把它 `\let` 成 `\relax`，则 `\ifx\<未定义>\@undefined` = `\ifx<空槽><原语槽>`
% → 类型不同 → **假**，分支翻转。
%
% 实测代价（§A1.quaterdecies）：regression-test.tex 的两处
% `\ifx\@@end\@undefined` 均翻成假 → `\let\@@@end\@@end`（undefined）+
% `\let\@@end\END` → `\END` 宏体反复重放 → 4986 轮后 input stack size=5000
% 爆栈（187 例官方用例中 111 例 STACK-END）。
%
% 故此处**刻意什么都不做**（让 `\@undefined` 保持未定义），仅留本注释说明原因。
% \CurrentFile / \@currname 等 LaTeX 文件名钩子（形参，实际未用）
\def\CurrentFile{}\def\@currname{}\def\@currext{}

% ── 载入官方 harness ────────────────────────────────────────────
% regression-test.tex 需从 l3build 包路径读入（由 -I 或 VFS 提供）
% ⚠ harness 由**用例自己** `\input{regression-test}` 载入，shim 不再代劳！
% 曾在此处载入 → harness 被**载入两次**（shim 一次 + 用例一次）→ 重入后的
% `\let\end\END`（regression-test.tex L87-91 的 BLOCK2 真分支）使 `\END` 宏
% 体末尾的 `\@@@end` 指向 `\END` 自身 → **无限自我递归** → 输入栈爆
% （187 例官方用例中 112 例 STACK-END，见 docs/expl3-lvt-scoreboard.md）。
% 官方 l3build 的驱动只用 `\input{testsuite}` 一次，shim 须保持一致语义。

% ── 载入 expl3 本体 ─────────────────────────────────────────────
% ⚠⚠ 这是「真分数」的前提（2026-09-11 实测，见 docs/expl3-real-workload.md）：
% 官方 `.lvt` 用例**假定 expl3 已由 LaTeX 内核装载**，直接使用
% `\cs_if_exist_use:N` 等函数。此前 shim 不载 expl3 → 用例里这些函数
% **全部 undefined**（实测 77 个/例），却因错误恢复跑到 `END-TEST-LOG`
% 被判 RAN —— 「RAN 180/187」是假分数（度量的是「能带着一堆 undefined
% 跑到末尾」）。载入 expl3 后同一用例 undefined 降到 6，真实差异才显现。
%
% 用 `expl3.ltx`（**不是** `expl3-code.tex`）：后者有 loader 检查
% （`\ifx\csname ExplLoaderFileDate\endcsname\relax` → `No expl3 loader detected`），
% `expl3.ltx` 首行 `\let\ExplLoaderFileDate\ExplFileDate` 提供该标志。
%
% ⚠ 不用 `\IfFileExists` / `\openin` 探测——**两者在 NTex 上均不可靠**
% （2026-09-11 实测：`\IfFileExists{expl3.ltx}` 走 false 分支却仍 `\input`；
% `\openin`+`\ifeof` 对存在的文件也报 eof）。改为**由 lvt-run.py 显式注入宏**
% `\lvtuseexplthree`（1 = 载入），职责清晰且不依赖未验证的原语。
% ⚠ 判据用 `\ifnum...=1`（lvt-run.py 总是先 `\chardef` 注入 0 或 1），
% 不用 `\ifx...\undefined`——NTex 上 undefined 哨兵语义有坑（见下文 §A1 注释）。
\ifnum\lvtuseexplthree=1
  \input expl3.ltx
\fi

% ── 跑目标测试 ──────────────────────────────────────────────────
\ifx\LVTFILE\undefined
  \message{[lvt-shim] 未定义 \string\LVTFILE}
\else
  % ⚠ 不可写 `\input{\LVTFILE}`：TeX 的 `\input` 是**文件名扫描**，不吃 `{...}`
  % 分组（`{}` 会被当文件名字符 → 「文件名含非法 token」）。须空格分隔。
  \input \LVTFILE
\fi

% ⚠ **不要在此加 `\end`**（2026-09-11 实测，§A1.undevicies）：
% shim 是被 `\input` 进来的，输入栈上还有外层帧。在**嵌套 input 上下文**里执行
% `\end` 会触发输入栈无限增长（实测：加 `\end` → `input stack size=5000` 爆栈；
% 去掉 → 正常）。作业终止由**最外层驱动文件**负责（`run.tex` 或 lvt-run.py 的
% driver），shim 只做定义 + 载入，不主动结束作业。
