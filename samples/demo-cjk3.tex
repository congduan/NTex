% demo-cjk3.tex —— M9 中文刀 4 端到端验收样张：中文粗体/黑体/楷体（家族伴随字体）
%
% 说明：
% - 刀 1/2 只有一款 FandolSong-Regular：`\bf` 只切拉丁字面（cmbx），汉字仍出
%   Regular 字形——标题/强调无粗体。本样张验收刀 4：FandolSong-Bold /
%   FandolHei-Regular / FandolKai-Regular 三款全链路（OTF 度量 → 折行 → DVI →
%   PDF CID 嵌入 + runtime CFF 子集化 → 文本层）。
% - **家族映射落在宏包层，不动引擎**：LaTeX 的 `\bfseries`/`\itshape` 走 NFSS
%   `\selectfont`（非 robust，`\gdef` 体），在其后补一步「按 series/shape 挑中文
%   伴随字体」。这与 ctex 的 `\bfseries`→黑体语义同向（ctexart 链未通前的可达
%   路径）；引擎级 family fallback（按码位段自动分派字体）是登记在
%   docs/KNOWN-SIMPLIFICATIONS.md 的未做项。
% - 中文字体经 `ntex_font::find_otf` 搜索链定位（NTEX_OTF_DIR → ~/.ntex-fonts →
%   系统字体目录 → texlive opentype 目录 → kpsewhich），Fandol 全家（GPL，
%   CTAN fonts/fandol）任一处存在即可。
% - 运行：
%     cargo run -p ntex-dvi -- samples/demo-cjk3.tex          # → samples/demo-cjk3.dvi（两趟 aux 闭合）
%     cargo run -p ntex-pdf -- samples/demo-cjk3.dvi          # → samples/demo-cjk3.pdf
%   验收：`pymupdf` 字体列表应出现 FandolSong-Bold / FandolHei-Regular /
%   FandolKai-Regular；第 2/3 行粗体字形目视加粗；文本层可提取。
\documentclass[12pt]{article}
\utfinputmode=1
\cjkbreakmode=1
% 常规/黑体两款显式给出；粗体与楷体由下面的桥按当前字号自动装载。
\font\zhf=FandolSong-Regular at 12pt
\font\zhhei=FandolHei-Regular at 14.4pt

% ---- 中文伴随字体桥（宏包层） ----
% 字体按需装载、以字号为键缓存（\f@size 进 csname）：\large/\small 语境下
% 伴随字体跟随 NFSS 字号，而不是钉死在定义时的固定尺寸。
\makeatletter
\g@addto@macro\selectfont{\ntex@cjk@companion}
\def\ntex@cjk@companion{%
  \edef\ntex@cjk@req{\f@series/\f@shape}%
  \def\ntex@cjk@tmp{bx/n}\ifx\ntex@cjk@req\ntex@cjk@tmp
    \ntex@cjk@load{b}{FandolSong-Bold}\else
  \def\ntex@cjk@tmp{b/n}\ifx\ntex@cjk@req\ntex@cjk@tmp
    \ntex@cjk@load{b}{FandolSong-Bold}\else
  \def\ntex@cjk@tmp{m/it}\ifx\ntex@cjk@req\ntex@cjk@tmp
    \ntex@cjk@load{k}{FandolKai-Regular}\else
  \def\ntex@cjk@tmp{bx/it}\ifx\ntex@cjk@req\ntex@cjk@tmp
    \ntex@cjk@load{k}{FandolKai-Regular}\else
  \def\ntex@cjk@tmp{b/it}\ifx\ntex@cjk@req\ntex@cjk@tmp
    \ntex@cjk@load{k}{FandolKai-Regular}\else
    \ntex@cjk@load{r}{FandolSong-Regular}%
  \fi\fi\fi\fi\fi}
% \ntex@cjk@load{<家族键>}{<字体文件名>}：按当前 \f@size 装载并缓存。
\def\ntex@cjk@load#1#2{%
  \expandafter\ifx\csname ntex@cjk@#1@\f@size\endcsname\relax
    \expandafter\font\csname ntex@cjk@#1@\f@size\endcsname=#2\space at \f@size pt\relax
  \fi
  \csname ntex@cjk@#1@\f@size\endcsname}
\makeatother

\begin{document}
\zhf

\par\vskip 6pt{\zhhei 中文粗体验收（刀 4）\par}\vskip 12pt

常规宋体：中文粗体验收探针。

{\bfseries 粗体宋体：中文粗体验收探针。}

\textbf{粗体命令：}行内粗体回到常规，数字 12345 与 Latin 保持 CM 字形。

{\itshape 楷体（斜体位）：中文没有真斜体，惯例以楷体承担。}

{\bfseries\itshape 粗斜体位同样落楷体（Fandol 无粗楷，粗宋楷不可兼得）。}

{\zhhei 黑体（标题层）：与粗宋互补，字形均匀无衬线。}

{\large\bfseries 字号跟随：本行是 larger 字号（14.4pt）的粗体，不是 12pt。\par}

\par\vskip 6pt{\zhhei 混排与断行\par}\vskip 8pt

\textbf{粗体中文}与常规中文在同一行内切换，Latin \textbf{bold} 与
\textbf{粗体} 同段混排：折行、字距、标点各走各的度量，互不串线。

\end{document}
